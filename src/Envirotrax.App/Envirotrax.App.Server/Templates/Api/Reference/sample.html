C# SELECT Request

using System.Net.Http;
using System.Xml;

public static async Task<XmlDocument> SendSampleSelectRequestAsync()
{
    var parameters = new Dictionary<string, string>
    {
        ["UserID"] = "api.user@example.com",
        ["ApiKey"] = "YOUR_API_KEY",
        ["RequestType"] = "SELECT",
        ["TableName"] = "Sites",
        ["SelectFields"] = "ID, CreationDate, PropertyType, PropertyBusinessName, PropertyStreetNumber, PropertyStreetName, PropertyNumber, PropertyCity, PropertyState, PropertyZIP",
        ["WhereLastModifiedDateTo"] = "9/1/2026",
        ["OrderBy"] = "ID",
        ["OrderByDirection"] = "ASC"
    };

    using var client = new HttpClient { Timeout = TimeSpan.FromMinutes(10) };
    using var content = new FormUrlEncodedContent(parameters);
    using var response = await client.PostAsync("/api/v1/run", content);

    var xml = await response.Content.ReadAsStringAsync();

    var document = new XmlDocument();
    document.LoadXml(xml);

    var status = document.SelectSingleNode("/EnvirotraxApiResponse/ResponseStatus")?.InnerText;

    if (status != "OK")
    {
        var message = document.SelectSingleNode("/EnvirotraxApiResponse/Errors/Error/ErrorMessage")?.InnerText;
        throw new InvalidOperationException(message);
    }

    return document;
}
